using UnityEngine; using UnityEngine.SceneManagement; using System.Collections; public class StickFightCompleteManager : MonoBehaviour { public static StickFightCompleteManager Instance; [Header("Match Settings")] [SerializeField] private float roundDuration = 20f; [SerializeField] private GameObject[] weaponPrefabs; [SerializeField] private Transform[] weaponSpawnPoints; [SerializeField] private float weaponSpawnInterval = 10f; private bool isMatchOver = false; private float weaponTimer; private void Awake() { if (Instance == null) Instance = this; else Destroy(gameObject); } private void Start() { StartCoroutine(RoundTimerRoutine()); } private void Update() { weaponTimer += Time.deltaTime; if (weaponTimer >= weaponSpawnInterval) { SpawnWeapon(); weaponTimer = 0f; } } private IEnumerator RoundTimerRoutine() { yield return new WaitForSeconds(roundDuration); if (!isMatchOver) { RestartLevel(); } } private void SpawnWeapon() { if (weaponPrefabs.Length == 0 || weaponSpawnPoints.Length == 0) return; int randomWeapon = Random.Range(0, weaponPrefabs.Length); int randomPoint = Random.Range(0, weaponSpawnPoints.Length); Instantiate(weaponPrefabs[randomWeapon], weaponSpawnPoints[randomPoint].position, Quaternion.identity); } public void NotifyPlayerDeath(GameObject deadPlayer) { if (isMatchOver) return; isMatchOver = true; Invoke("RestartLevel", 2f); } private void RestartLevel() { SceneManager.LoadScene(SceneManager.GetActiveScene().name); } } [RequireComponent(typeof(Rigidbody2D))] public class StickPlayerController : MonoBehaviour { [SerializeField] private float moveSpeed = 12f; [SerializeField] private float jumpForce = 16f; [SerializeField] private LayerMask groundLayer; [SerializeField] private Transform groundCheck; [SerializeField] private Rigidbody2D[] bodyParts; private Rigidbody2D rb; private float horizontalInput; private bool isGrounded; private void Awake() { rb = GetComponent(); } private void Update() { horizontalInput = Input.GetAxisRaw("Horizontal"); isGrounded = Physics2D.OverlapCircle(groundCheck.position, 0.2f, groundLayer); if ((Input.GetButtonDown("Jump") || Input.GetKeyDown(KeyCode.W)) && isGrounded) { rb.velocity = new Vector2(rb.velocity.x, jumpForce); } } private void FixedUpdate() { rb.velocity = new Vector2(horizontalInput * moveSpeed, rb.velocity.y); if (horizontalInput != 0) { transform.localScale = new Vector3(Mathf.Sign(horizontalInput), 1, 1); } } public void EnableFullRagdoll(bool enable) { foreach (var part in bodyParts) { part.isKinematic = !enable; } } private void OnCollisionEnter2D(Collision2D collision) { if (collision.relativeVelocity.magnitude > 25f) { EnableFullRagdoll(true); if (StickFightCompleteManager.Instance != null) { StickFightCompleteManager.Instance.NotifyPlayerDeath(gameObject); } } } } public class WeaponSystem : MonoBehaviour { [SerializeField] private GameObject bulletPrefab; [SerializeField] private Transform firePoint; [SerializeField] private float fireRate = 0.1f; [SerializeField] private float bulletForce = 25f; [SerializeField] private float recoilForce = 5f; private float nextFireTime; private Rigidbody2D playerRb; private void Start() { playerRb = GetComponentInParent(); } public void Shoot() { if (Time.time < nextFireTime) return; nextFireTime = Time.time + fireRate; GameObject bullet = Instantiate(bulletPrefab, firePoint.position, firePoint.rotation); Rigidbody2D bulletRb = bullet.GetComponent(); bulletRb.AddForce(firePoint.right * bulletForce, ForceMode2D.Impulse); if (playerRb != null) { playerRb.AddForce(-firePoint.right * recoilForce, ForceMode2D.Impulse); } } }
